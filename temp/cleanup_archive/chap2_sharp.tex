\prop{2.1}{d}{Eit \textbf{seleksjonstrykk} endrar sannsynsfordelinga i formrommet.\footnote{Wright (1932)}\footnote{Kostnad, latens og minne deformerer topologien.}}
\prop{2.11}{t}{Trykket er inga regel, men ein gradient som aukar sannsynet for visse posisjonar.\footnote{Gradienten tvingar fram mønsteret utan å diktere kodelinja.}}

\prop{2.1111}{t}{Trykket avgrensar kvar posisjonen ikkje kan vere.\footnote{Feilhandtering definerer yttergrensene for utføringsstien.}}
\prop{2.11111}{t}{Forma står att når trykka har utelukka alt anna.\footnote{Arkitekturen er restverdien når krava har fjerna dei umoglege alternativa.}}
\prop{2.11112}{t}{Forma er ein rest.\footnote{Systemutvikling er reduktiv feilretting.}}

\prop{2.111122}{t}{Eit trykk som utelukkar alt unnateke éin region, er determinisme.\footnote{Eit system med éin tilstand ber inga informasjon.}}
\prop{2.111123}{t}{Observerte trykk ligg mellom desse ytterpunkta.\footnote{Heuristisk søk opererer mellom fridom og tvang.}}
\prop{2.12}{a}{Fleire seleksjonstrykk verkar samstundes i kvar funksjonell klasse.\footnote{Konsensus og tilgjenge utgjer kryssande trykk i eit distribuert system.}}
\prop{2.121}{t}{Einsidige formforklaringar er ufullstendige.\footnote{Suksess skuldast aldri berre eitt trykk.}}
\prop{2.1211}{t}{Med berre eitt trykk ville alle former vore identiske eller tilfeldig fordelte langs éin akse.\footnote{Eit nevralt nettverk med éin tapsfunksjon kollapsar.}}
\prop{2.12111}{t}{Ikkje-uniformiteten krev fleire uavhengige trykk.\footnote{Spreiing i kodelager vitnar om motstridande avhengnader.}}

\prop{2.12121}{o}{Uvarierte trykk lèt seg knapt telje.\footnote{Nettverksprotokollar er usynlege inntil pakketapet stig.}}
\prop{2.121211}{t}{Metoden krev variasjon i vilkåra.\footnote{Stresstesting krev partisjonering, ikkje berre last.}}
\prop{2.13}{a}{Minst to trykk peikar i motstridande retningar i minst éin region.\footnote{CAP-teoremet formaliserer motsegna mellom konsistens og tilgjenge.}}
\prop{2.131}{t}{Trykk i motstridande retningar utelukkar globale optimum.\footnote{Ingen database gjev både maksimal normalisering og minimal leselatens.}}
\prop{2.1311}{t}{Kvar realisert form er eit kompromiss.\footnote{Michl (1995)}\footnote{Mikrotenester ofrar bandbreidde for isolasjon.}}
\prop{2.13111}{t}{Kompromisset er den einaste moglege balansen.\footnote{Eventuell konsistens er ei naudsynt tilpassing.}}
\prop{2.13112}{t}{Formskilnader syner at kreftene lèt seg balansere ulikt.\footnote{Monolittar og mikrokjernar balanserer same krefter ulikt.}}
\prop{2.13113}{t}{Omgrepet «suboptimal» føreset ein referanse.\footnote{Ein $O(n^2)$-algoritme er optimal når minne trumfar tid.}}
\prop{2.131131}{t}{Utan eit spesifikt trykksett er «suboptimal» ikkje falsifiserbart.\footnote{Vurderinga manglar form når ho ikkje vekter yting mot lesbarheit.}}
\prop{2.14}{t}{Funksjonen definerer klassen og formrommet.\footnote{API-kontrakten stakar ut integrasjonsrommet.}}

\prop{2.141111}{i}{Stolen og sofaen utgjer ulike klasser med ulik funksjon.\footnote{Køyretydsmiljøet og kompilatoren opererer i ulike tidsdomene.}}
