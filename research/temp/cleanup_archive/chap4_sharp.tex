\silentchapter{4 Landskapet er dynamisk}{landskapet er dynamisk}
\prop{4}{}{Landskapet er dynamisk.}
\prop{4.1}{a}{Seleksjonstrykka kviler aldri.\footnote{Teknisk gjeld og nye krav tvingar trykka til å fluktuere. Krav til responstid og datakonsistens omformar domenet tvert.}}
\prop{4.11}{t}{Av 3.1 og 4.1: Landskapet er ei uroleg flate.\footnote{Nettverkstopologien er ustabil; nodar døyr, bandbreidda snevrar inn.}}
\prop{4.111}{t}{Haugar stig, søkk, flyttar seg, deler seg eller smeltar saman.}
\prop{4.1111}{t}{Ein optimal posisjon vert suboptimal straks vilkåra endrar seg.\footnote{Ein arkitektur optimal for tusen brukarar sviktar for ein million. Tilpassinga er lokal i rom og skala. Synkrone paradigme knekk under asynkron last.}}
\prop{4.11111}{t}{Optimal tilpassing er lokal i tid.}
\prop{4.11112}{o}{Ein ny teknologi flyttar grensa for det forbodne.\footnote{Nye abstraksjonslag opnar låste regionar av kompleksitet. Distribuert konsensus gjer umoglege transaksjonar trivielle.}}
\prop{4.111121}{o}{Ein ny marknad endrar kva landskapet belønnar.\footnote{Mobil brukaråtferd straffar informasjonstettleik og premierer minimalisme.}}
\prop{4.111122}{o}{Eit nytt materiale utvidar operasjonsrommet.\footnote{Grafbaserte datastrukturar opnar for heilt nye oppslagstypologiar.}}
\prop{4.111123}{t}{Kvart slikt skifte grip inn i landskapet.}
\prop{4.11113}{o}{Medianen for proporsjonen høgde til breidde fell systematisk over fem hundreår.}
\prop{4.111131}{o}{Proporsjonen 1.36 er ikkje meir ergonomisk enn 1.88; begge høver for menneskekroppen.}
\prop{4.111132}{o}{Dette er formdrift, ikkje funksjonsdrift.}
\prop{4.2}{o}{Endringa er diskontinuerleg.\footnote{Eldredge \& Gould (1972)\footnote{Paradigmeskifte bryt stasen med brå topologiske restruktureringar. Overgangen frå monolitt til mikrotjenester slettar det gamle kartet umiddelbart.}}}
\prop{4.21}{o}{Lange periodar med stase bryt saman i brå topologiske skifte.\footnote{Heile klassar av designmønster døyr over natta når køyremiljøet vert oppdatert.}}
\prop{4.211}{o}{Forløpet er alltid: eksplosiv radiasjon inn i ein ny region, deretter konvergens mot nye attraktorar.\footnote{Nye bibliotek utløyser ein eksplosjon av løysingar før dei sterkaste mønstera standardiserast.}}
\prop{4.2111}{t}{Episodisk diskontinuitet bryt ikkje med den underliggande dynamikken.}
\prop{4.21111}{t}{Ho er signaturen til denne dynamikken.}
\prop{4.21112}{o}{At endringsraten er diskontinuerleg, og ikkje bimodal, stadfestar dette.}
\prop{4.21113}{o}{Brotet etter 1900 syner ei rekurrensmatrise der ingen periode før 1900 liknar nokon etter.}
\prop{4.211131}{o}{Formhistoria gjentek seg aldri over dette brotet.}
\prop{4.3}{t}{Landskapet har minne.\footnote{Arthur (1994)\footnote{Eksisterande databasestrukturar avgrensar nye tenester. Tidlegare avgjerder ligg innbakte i koden som spøkelsestrykk.}}}
\prop{4.31}{o}{Kvar realisert form set spor som endrar seleksjonstrykka for den neste.\footnote{Eit publisert API bind framtida; å fjerne eit endepunkt knuser økosystemet rundt det.}}
\prop{4.311}{t}{Form og landskap endrar kvarandre.}
\prop{4.3111}{t}{All formgjeving er omformgjeving; agenten startar aldri med blanke ark.\footnote{Rammeverket ber allereie stiavhengige seleksjonstrykk. Gammal kode dikterer den nye.}}
\prop{4.31111}{t}{Å starte frå ein posisjon er å arve alle implikasjonane hennar.}
\prop{4.311111}{t}{Agenten arvar både dei opne og dei stengde naboregionane.}
\prop{4.311112}{t}{Utgangspunktet er aldri nøytralt.}
\prop{4.31112}{t}{Stiavhengigheit er eit fundamentalt trekk ved formhistoria, ingen anomali.\footnote{Asynkrone paradigme låser systemet slik at synkrone mønster vert utilgjengelege utan massiv refaktorering. Arven stenger dører.}}
\prop{4.311121}{o}{At mahogni okkuperte 16 av 16 stolar i 1825–1849, etter totalt fråvær i perioden før, er klår stiavhengig seleksjonskollaps.}
\prop{4.4}{o}{Formrommet ekspanderer kumulativt.\footnote{API-økosystem veks monotont; å fjerne funksjonar skjuler berre kompleksiteten medan konseptrommet utvidast.}}
\prop{4.41}{o}{Nye regionar vert okkuperte; dei gamle vert sjeldan heilt forlatne.\footnote{Gamle protokollar må haldast ved like for å sikre bakoverkompatibilitet. Historiske lag eksisterer samstundes.}}
\prop{4.411}{o}{Det kumulative konvekse hylstervolumet veks monotont gjennom alle periodar, utan eitt einaste tilbakesteg.}
\prop{4.4111}{t}{Eit fast funksjonsenvelop tvingar fram metning.}
\prop{4.41111}{o}{Nyheitsraten fell ikkje monotont mot metning; han skyt fart att når nye materiale trer inn.}
\prop{4.41112}{t}{Det tilstøytande moglege ekspanderer aktivt; det er inkje lager som tømmest.\footnote{Kauffman (1993)\footnote{Å opne kjeldekoden tømmer ikkje rommet, men gjer eksponentielt fleire posisjonar tilgjengelege for agentane.}}}
\prop{4.41113}{t}{Materialstraumar driv landskapsendringa.}
\prop{4.411131}{o}{Kvart nytt materiale opnar nokre regionar og låser andre.\footnote{Distribuert logikk mogleggjer global skala, men knuser atomiske tilstandsgarantiar.}}
\prop{4.411132}{o}{Materialstraumen frå 1500 til 2025 syner tydelege seleksjonsbølgjer.}
\prop{4.411133}{o}{Kvar bølgje er ei brå kohorthending, ingen gradvis overgang mot overlegne alternativ.}
\prop{4.5}{o}{Når eitt seleksjonstrykk dominerer, kollapsar landskapet mot éin attraktor.\footnote{Viss fart er einaste metrikk, stivnar arkitekturen. Eit absolutt krav om tryggleik frys systemet fast.}}
\prop{4.51}{t}{Eit eineveldande trykk forklarer ikkje form.}
\prop{4.511}{t}{Det er fråværet av forklaring.}
\prop{4.5111}{t}{Formrommets diversitet speglar mangfaldet av aktive trykk.\footnote{Robust nettverk krev balanse mellom latens, gjennomstrøyming og feiltoleranse. Dominans av éin parameter drep systemet.}}
\prop{4.51111}{t}{Ein monokultur i formrommet provar at eitt trykk har vorte eineveldande.\footnote{Kloning av brukargrensesnitt syner at frykta for kognitiv friksjon overkøyrer all estetikk.}}
